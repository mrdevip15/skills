    // Programmatic Slicing & Resizing Pipeline
const sharp = require('sharp');
const manifest = require('./icon_grid_manifest.json');

async function sliceAndExport() {
  for (const [sheetKey, sheet] of Object.entries(manifest.sheets)) {
    const atlasImg = sharp(sheet.source);
    
    for (const icon of sheet.icons) {
      await sharp(sheet.source)
        .extract({
          left: icon.x,
          top: icon.y,
          width: icon.width,
          height: icon.height
        })
        .resize(sheet.final_icon_size, sheet.final_icon_size, {
          kernel: sharp.kernel.nearest // Crucial for pixel art
        })
        .png()
        .toFile(icon.output);
        
      console.log(`✓ Sliced & saved [${sheet.final_icon_size}x${sheet.final_icon_size}]: ${icon.output}`);
    }
  }
}
sliceAndExport();
